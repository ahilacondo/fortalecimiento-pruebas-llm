package org.apache.commons.codec.binary;

import static org.junit.Assert.assertArrayEquals;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.fail;

import java.math.BigInteger;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.Random;

import org.apache.commons.codec.DecoderException;
import org.apache.commons.codec.EncoderException;
import org.junit.Test;

/**
 * Pruebas adicionales para {@link Base64} que usan únicamente la API pública.
 */
public class Base64LLMTest {

    private static final byte[] CRLF = {'\r', '\n'};

    private static byte[] ascii(String s) {
        return s.getBytes(StandardCharsets.US_ASCII);
    }

    private static String str(byte[] b) {
        return new String(b, StandardCharsets.US_ASCII);
    }

    private static byte[] randomBytes(Random rnd, int n) {
        byte[] data = new byte[n];
        rnd.nextBytes(data);
        return data;
    }

    private static void assertLinesWithin76(byte[] enc) {
        int count = 0;
        for (int i = 0; i < enc.length; i++) {
            if (enc[i] == '\r') {
                assertTrue("CR debe ir seguido de LF", i + 1 < enc.length && enc[i + 1] == '\n');
                assertTrue("Línea mayor de 76 caracteres", count <= 76);
                count = 0;
                i++;
            } else {
                count++;
                assertTrue("Línea mayor de 76 caracteres", count <= 76);
            }
        }
    }

    // ---------------------------------------------------------------- nulos y vacíos

    @Test
    public void testNullInputsReturnNull() {
        assertNull(Base64.encodeBase64(null));
        assertNull(Base64.encodeBase64(null, true));
        assertNull(Base64.encodeBase64(null, false, true));
        assertNull(Base64.decodeBase64(null));
    }

    @Test
    public void testEmptyInputsReturnEmpty() {
        assertEquals(0, Base64.encodeBase64(new byte[0], true, true).length);
        assertEquals(0, Base64.encodeBase64URLSafe(new byte[0]).length);
        assertEquals(0, Base64.decodeBase64(new byte[0]).length);
    }

    // ---------------------------------------------------------------- codificación conocida

    @Test
    public void testStandardEncodingOfTwoBytesUsesPlusSlashAndPadding() {
        assertEquals("+/8=", str(Base64.encodeBase64(new byte[] {(byte) 0xfb, (byte) 0xff})));
    }

    @Test
    public void testUrlSafeEncodingOfTwoBytesUsesDashUnderscoreAndNoPadding() {
        assertEquals("-_8", str(Base64.encodeBase64URLSafe(new byte[] {(byte) 0xfb, (byte) 0xff})));
    }

    @Test
    public void testSingleByteFBEncodings() {
        assertEquals("+w==", str(Base64.encodeBase64(new byte[] {(byte) 0xfb})));
        assertEquals("-w", str(Base64.encodeBase64URLSafe(new byte[] {(byte) 0xfb})));
    }

    @Test
    public void testAllOnesEncodings() {
        byte[] ones = {(byte) 0xff, (byte) 0xff, (byte) 0xff};
        assertEquals("////", str(Base64.encodeBase64(ones)));
        assertEquals("____", str(Base64.encodeBase64URLSafe(ones)));
    }

    @Test
    public void testEncodeBase64WithUrlSafeFlagMatchesDedicatedMethods() {
        Random rnd = new Random(1001);
        for (int n = 0; n < 40; n++) {
            byte[] data = randomBytes(rnd, n);
            assertArrayEquals(Base64.encodeBase64(data), Base64.encodeBase64(data, false, false));
            assertArrayEquals(Base64.encodeBase64URLSafe(data), Base64.encodeBase64(data, false, true));
            assertArrayEquals(Base64.encodeBase64Chunked(data), Base64.encodeBase64(data, true, false));
        }
    }

    @Test
    public void testUrlSafeChunkedOutputIsTrimmedWithoutPadding() {
        byte[] enc = Base64.encodeBase64(new byte[] {(byte) 0xfb, (byte) 0xff}, true, true);
        assertEquals("-_8\r\n", str(enc));
        assertEquals(5, enc.length);
    }

    @Test
    public void testUrlSafeOutputNeverContainsStandardOnlyCharacters() {
        Random rnd = new Random(2002);
        for (int i = 0; i < 50; i++) {
            byte[] enc = Base64.encodeBase64URLSafe(randomBytes(rnd, 1 + rnd.nextInt(300)));
            for (byte b : enc) {
                assertTrue('+' != b);
                assertTrue('/' != b);
                assertTrue('=' != b);
            }
        }
    }

    @Test
    public void testUnpaddedUrlSafeLengthMatchesFormula() {
        Random rnd = new Random(3003);
        for (int n = 1; n <= 30; n++) {
            byte[] enc = Base64.encodeBase64URLSafe(randomBytes(rnd, n));
            assertEquals((4 * n + 2) / 3, enc.length);
        }
    }

    @Test
    public void testPaddedStandardLengthIsMultipleOfFour() {
        for (int n = 1; n <= 30; n++) {
            byte[] enc = Base64.encodeBase64(new byte[n]);
            assertEquals(0, enc.length % 4);
            assertEquals(4 * ((n + 2) / 3), enc.length);
        }
    }

    // ---------------------------------------------------------------- instancia y modo URL-safe

    @Test
    public void testIsUrlSafeReflectsConstructorMode() {
        assertTrue(new Base64(true).isUrlSafe());
        assertFalse(new Base64(false).isUrlSafe());
        assertFalse(new Base64().isUrlSafe());
        assertTrue(new Base64(76, CRLF, true).isUrlSafe());
        assertFalse(new Base64(76, CRLF, false).isUrlSafe());
    }

    @Test
    public void testInstanceEncodeHonorsUrlSafeMode() {
        byte[] data = {(byte) 0xfb, (byte) 0xff};
        assertEquals("-_8", str(new Base64(true).encode(data)));
        assertEquals("+/8=", str(new Base64().encode(data)));
    }

    @Test
    public void testInstanceDecodeArray() {
        assertEquals("Hello", str(new Base64().decode(ascii("SGVsbG8="))));
    }

    @Test
    public void testEncodeObjectWithByteArray() throws EncoderException {
        Object result = new Base64().encode((Object) ascii("Hello"));
        assertTrue(result instanceof byte[]);
        assertEquals("SGVsbG8=", str((byte[]) result));
    }

    @Test
    public void testEncodeObjectRejectsNonByteArray() {
        try {
            new Base64().encode((Object) "Hello");
            fail("Se esperaba EncoderException");
        } catch (EncoderException expected) {
            // esperado
        }
    }

    @Test
    public void testDecodeObjectWithByteArray() throws DecoderException {
        Object result = new Base64().decode((Object) ascii("SGVsbG8="));
        assertTrue(result instanceof byte[]);
        assertEquals("Hello", str((byte[]) result));
    }

    @Test
    public void testDecodeObjectRejectsNonByteArray() {
        try {
            new Base64().decode((Object) "SGVsbG8=");
            fail("Se esperaba DecoderException");
        } catch (DecoderException expected) {
            // esperado
        }
    }

    // ---------------------------------------------------------------- decodificación

    @Test
    public void testDecodeUnpaddedInputs() {
        assertEquals("AB", str(Base64.decodeBase64(ascii("QUI"))));
        assertEquals("A", str(Base64.decodeBase64(ascii("QQ"))));
        assertEquals("ABC", str(Base64.decodeBase64(ascii("QUJDR"))));
    }

    @Test
    public void testDecodeStopsAtFirstPadding() {
        assertEquals("A", str(Base64.decodeBase64(ascii("QQ==QUI="))));
    }

    @Test
    public void testDecodeIgnoresLineBreaksAndWhitespace() {
        assertEquals("ABCDEF", str(Base64.decodeBase64(ascii("QUJD\r\nREVG"))));
        assertEquals("ABCDEF", str(Base64.decodeBase64(ascii("QUJD REVG\t"))));
    }

    @Test
    public void testDecodeAcceptsBothAlphabets() {
        byte[] expected = {(byte) 0xfb, (byte) 0xff};
        assertArrayEquals(expected, Base64.decodeBase64(ascii("+/8=")));
        assertArrayEquals(expected, Base64.decodeBase64(ascii("-_8=")));
        assertArrayEquals(expected, Base64.decodeBase64(ascii("-_8")));
    }

    @Test
    public void testDecodeRoundTripUrlSafeAndStandard() {
        Random rnd = new Random(4004);
        for (int n = 0; n < 60; n++) {
            byte[] data = randomBytes(rnd, n);
            assertArrayEquals(data, Base64.decodeBase64(Base64.encodeBase64(data)));
            assertArrayEquals(data, Base64.decodeBase64(Base64.encodeBase64URLSafe(data)));
        }
    }

    @Test
    public void testDecodeLargeRandomRoundTrip() {
        Random rnd = new Random(5005);
        byte[] data = randomBytes(rnd, 20000);
        assertArrayEquals(data, Base64.decodeBase64(Base64.encodeBase64(data)));
        assertArrayEquals(data, Base64.decodeBase64(Base64.encodeBase64URLSafe(data)));
        assertArrayEquals(data, Base64.decodeBase64(Base64.encodeBase64Chunked(data)));
    }

    // ---------------------------------------------------------------- chunked

    @Test
    public void testChunkedLinesNeverExceed76Characters() {
        Random rnd = new Random(6006);
        for (int n = 0; n <= 150; n++) {
            assertLinesWithin76(Base64.encodeBase64Chunked(randomBytes(rnd, n)));
        }
    }

    @Test
    public void testChunkedRoundTripLargeInput() {
        Random rnd = new Random(7007);
        byte[] data = randomBytes(rnd, 10000);
        byte[] enc = Base64.encodeBase64Chunked(data);
        assertLinesWithin76(enc);
        assertArrayEquals(data, Base64.decodeBase64(enc));
    }
