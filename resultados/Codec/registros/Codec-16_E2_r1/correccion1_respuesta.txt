    private static byte[] randomBytes(final long seed, final int length) {
        final Random random = new Random(seed);
        final byte[] data = new byte[length];
        random.nextBytes(data);
        return data;

        final byte[] full = bytes("foobar");
        final String encoded = codec.encodeAsString(full);
        for (int chars = 1; chars <= 7; chars++) {
            final String truncated = encoded.substring(0, chars).replace("=", "");
            final byte[] decoded = codec.decode(truncated);
            final int expectedBytes = (chars * 5) / 8;
