    private static byte[] remainingBytes(final ByteBuffer buffer) {
        final ByteBuffer copy = buffer.duplicate();
        final byte[] out = new byte[copy.remaining()];
        copy.get(out);
        return out;
    }

    @Test
    public void testConcurrentStaticCallsReturnConsistentResults() throws Exception {
        final int threadCount = 8;
        final int iterations = 500;
        final byte[] expectedUtf8 = NON_ASCII.getBytes(StandardCharsets.UTF_8);
        final byte[] expectedUtf16Le = NON_ASCII.getBytes(StandardCharsets.UTF_16LE);

        final ExecutorService pool = Executors.newFixedThreadPool(threadCount);
        try {
            final List<Future<Boolean>> futures = new ArrayList<>();
            for (int t = 0; t < threadCount; t++) {
                futures.add(pool.submit(() -> {
                    for (int i = 0; i < iterations; i++) {
                        Assert.assertTrue(StringUtils.equals("abc", "abc"));
                        Assert.assertFalse(StringUtils.equals("abc", "ABC"));
                        Assert.assertTrue(StringUtils.equals(null, null));
                        Assert.assertTrue(StringUtils.equals(new StringBuilder(NON_ASCII), NON_ASCII));
                        Assert.assertArrayEquals(expectedUtf8, StringUtils.getBytesUtf8(NON_ASCII));
                        Assert.assertEquals(NON_ASCII, StringUtils.newStringUtf8(expectedUtf8));
                        Assert.assertArrayEquals(expectedUtf16Le, StringUtils.getBytesUtf16Le(NON_ASCII));
                        Assert.assertEquals(NON_ASCII, StringUtils.newStringUtf16Le(expectedUtf16Le));
                        Assert.assertArrayEquals(expectedUtf8,
                                remainingBytes(StringUtils.getByteBufferUtf8(NON_ASCII)));
                    }
                    return Boolean.TRUE;
                }));
            }
            for (final Future<Boolean> future : futures) {
                // Si un hilo falla, get() lanza ExecutionException con la AssertionError como causa.
                Assert.assertTrue(future.get(60, TimeUnit.SECONDS));
            }
        } finally {
            pool.shutdownNow();
        }
    }
