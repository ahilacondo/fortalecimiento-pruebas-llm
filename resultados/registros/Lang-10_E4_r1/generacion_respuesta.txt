    private static void assertSameAsSimpleDateFormat(String pattern, String input) throws ParseException {
        SimpleDateFormat sdf = new SimpleDateFormat(pattern, Locale.US);
        sdf.setTimeZone(GMT);
        Date expected = sdf.parse(input);
        Date actual = newParser(pattern, GMT, Locale.US).parse(input);
        assertEquals("Patrón [" + pattern + "] entrada [" + input + "]", expected, actual);
    }

    private static void assertDeserializedBehavesLikeOriginal(String pattern, TimeZone tz,
            Locale locale, String input) throws ParseException {
        FastDateParser original = newParser(pattern, tz, locale);
        FastDateParser copy = SerializationUtils.deserialize(SerializationUtils.serialize(original));
        assertEquals("Patrón [" + pattern + "] entrada [" + input + "]",
                original.parse(input), copy.parse(input));
    }

    // 1. Thread-safety

    @Test
    public void testSharedInstanceParsesCorrectlyFromConcurrentThreads() throws Exception {
        final FastDateParser parser = newParser("yyyy-MM-dd HH:mm", GMT, Locale.US);
        final int n = 200;
        final String[] inputs = new String[n];
        final Date[] expected = new Date[n];
        for (int i = 0; i < n; i++) {
            int year = 2000 + (i % 20);
            int month = 1 + (i % 12);
            int day = 1 + (i % 28);
            int hour = i % 24;
            int minute = (i * 7) % 60;
            inputs[i] = String.format(Locale.US, "%04d-%02d-%02d %02d:%02d",
                    year, month, day, hour, minute);
            Calendar cal = Calendar.getInstance(GMT);
            cal.clear();
            cal.set(year, month - 1, day, hour, minute);
            expected[i] = cal.getTime();
        }

        final int threads = 8;
        ExecutorService pool = Executors.newFixedThreadPool(threads);
        try {
            final CountDownLatch start = new CountDownLatch(1);
            List<Future<String>> futures = new ArrayList<>();
            for (int t = 0; t < threads; t++) {
                final int offset = t;
                futures.add(pool.submit(() -> {
                    start.await();
                    for (int k = 0; k < n; k++) {
                        int i = (k + offset * 31) % n;
                        Date actual = parser.parse(inputs[i]);
                        if (!expected[i].equals(actual)) {
                            return "Hilo " + offset + ": entrada " + inputs[i]
                                    + " esperaba " + expected[i] + " y obtuvo " + actual;
                        }
                    }
                    return null;
                }));
            }
            start.countDown();
            for (Future<String> f : futures) {
                String error = f.get(60, TimeUnit.SECONDS);
                assertNull(error, error);
            }
        } finally {
            pool.shutdownNow();
        }
    }
