package org.apache.commons.cli;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNotSame;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.fail;

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.io.PrintWriter;
import java.io.StringWriter;
import java.util.Comparator;
import java.util.function.Consumer;

import org.junit.Test;

/**
 * Pruebas adicionales para {@link HelpFormatter} que cubren requisitos de la
 * documentación no verificados por la suite existente. Solo usan la API pública.
 */
public class HelpFormatterLLMTest
{
    private static final String EOL = System.getProperty("line.separator");

    /** Ejecuta una acción sobre un PrintWriter respaldado por un StringWriter y devuelve el contenido. */
    private static String run(Consumer<PrintWriter> action)
    {
        StringWriter sw = new StringWriter();
        PrintWriter pw = new PrintWriter(sw);
        action.accept(pw);
        pw.flush();
        return sw.toString();
    }

    /** Captura lo que se escribe en System.out durante la ejecución de la acción. */
    private static String captureStdout(Runnable action)
    {
        PrintStream original = System.out;
        ByteArrayOutputStream buffer = new ByteArrayOutputStream();
        System.setOut(new PrintStream(buffer, true));
        try
        {
            action.run();
        }
        finally
        {
            System.setOut(original);
        }
        return buffer.toString();
    }

    // ---------------------------------------------------------------- Req. 1

    @Test
    public void testPublicConstantsHaveDocumentedValues()
    {
        assertEquals(74, HelpFormatter.DEFAULT_WIDTH);
        assertEquals(1, HelpFormatter.DEFAULT_LEFT_PAD);
        assertEquals(3, HelpFormatter.DEFAULT_DESC_PAD);
        assertEquals("usage: ", HelpFormatter.DEFAULT_SYNTAX_PREFIX);
        assertEquals("-", HelpFormatter.DEFAULT_OPT_PREFIX);
        assertEquals("--", HelpFormatter.DEFAULT_LONG_OPT_PREFIX);
        assertEquals("arg", HelpFormatter.DEFAULT_ARG_NAME);
    }

    @Test
    public void testDeprecatedFieldsAreInitializedWithDefaults()
    {
        HelpFormatter hf = new HelpFormatter();

        assertEquals(HelpFormatter.DEFAULT_WIDTH, hf.defaultWidth);
        assertEquals(HelpFormatter.DEFAULT_LEFT_PAD, hf.defaultLeftPad);
        assertEquals(HelpFormatter.DEFAULT_DESC_PAD, hf.defaultDescPad);
        assertEquals(HelpFormatter.DEFAULT_SYNTAX_PREFIX, hf.defaultSyntaxPrefix);
        assertEquals(EOL, hf.defaultNewLine);
        assertEquals(HelpFormatter.DEFAULT_OPT_PREFIX, hf.defaultOptPrefix);
        assertEquals(HelpFormatter.DEFAULT_LONG_OPT_PREFIX, hf.defaultLongOptPrefix);
        assertEquals(HelpFormatter.DEFAULT_ARG_NAME, hf.defaultArgName);
    }

    @Test
    public void testGettersReturnDefaultValues()
    {
        HelpFormatter hf = new HelpFormatter();

        assertEquals(74, hf.getWidth());
        assertEquals(1, hf.getLeftPadding());
        assertEquals(3, hf.getDescPadding());
        assertEquals("usage: ", hf.getSyntaxPrefix());
        assertEquals(EOL, hf.getNewLine());
        assertEquals("-", hf.getOptPrefix());
        assertEquals("--", hf.getLongOptPrefix());
        assertEquals("arg", hf.getArgName());
    }

    // ---------------------------------------------------------------- Req. 2

    @Test
    public void testPrintHelpWithoutAutoUsageWritesToSystemOut()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = captureStdout(() -> hf.printHelp("app", options));

        assertEquals("usage: app" + EOL + " -a   first" + EOL, out);
    }

    @Test
    public void testPrintHelpWithAutoUsageWritesGeneratedUsageToSystemOut()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = captureStdout(() -> hf.printHelp("app", options, true));

        assertEquals("usage: app [-a]" + EOL + " -a   first" + EOL, out);
    }

    // ---------------------------------------------------------------- Req. 3

    @Test
    public void testPrintHelpWithWidthWrapsDescriptionAtGivenWidth()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "aaaa bbbb cccc dddd eeee ffff");

        String out = captureStdout(() -> hf.printHelp(30, "app", null, options, null));

        assertEquals("usage: app" + EOL
                + " -a   aaaa bbbb cccc dddd eeee" + EOL
                + "      ffff" + EOL, out);
        for (String line : out.split("\\R"))
        {
            assertTrue("line exceeds width: '" + line + "'", line.length() <= 30);
        }
    }

    // ---------------------------------------------------------------- Req. 4

    @Test
    public void testNullOrBlankHeaderAndFooterAreNotPrinted()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String plain = run(pw -> hf.printHelp(pw, 80, "app", null, options, 1, 3, null, false));
        String blank = run(pw -> hf.printHelp(pw, 80, "app", "   ", options, 1, 3, "\t \n", false));

        assertEquals(plain, blank);
        assertEquals("usage: app" + EOL + " -a   first" + EOL, plain);
    }

    @Test
    public void testNonBlankHeaderAndFooterArePrinted()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = run(pw -> hf.printHelp(pw, 80, "app", "Header", options, 1, 3, "Footer", false));

        assertEquals("usage: app" + EOL
                + "Header" + EOL
                + " -a   first" + EOL
                + "Footer" + EOL, out);
    }

    // ---------------------------------------------------------------- Req. 5

    @Test
    public void testSyntaxIsUsedLiterallyWhenAutoUsageIsFalse()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = run(pw -> hf.printHelp(pw, 80, "app -x <file>", null, options, 1, 3, null, false));

        assertEquals("usage: app -x <file>" + EOL + " -a   first" + EOL, out);
        assertFalse(out.contains("[-a]"));
    }

    @Test
    public void testAutoUsageAppendsGeneratedOptionsToSyntax()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = run(pw -> hf.printHelp(pw, 80, "app -x <file>", null, options, 1, 3, null, true));

        assertEquals("usage: app -x <file> [-a]" + EOL + " -a   first" + EOL, out);
    }

    // ---------------------------------------------------------------- Req. 6

    @Test
    public void testPrintHelpThrowsWhenNoRoomForDescription()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "aaaa bbbb cccc");

        try
        {
            hf.printHelp(new PrintWriter(new StringWriter()), 6, "app", null, options, 1, 3, null, false);
            fail("IllegalStateException expected when width is not greater than the indentation");
        }
        catch (IllegalStateException expected)
        {
            // expected
        }
    }

    @Test
    public void testPrintOptionsThrowsWhenNoRoomForDescription()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "aaaa bbbb cccc");

        try
        {
            hf.printOptions(new PrintWriter(new StringWriter()), 6, options, 1, 3);
            fail("IllegalStateException expected when width is not greater than the indentation");
        }
        catch (IllegalStateException expected)
        {
            // expected
        }
    }

    // ---------------------------------------------------------------- Req. 7

    @Test
    public void testDefaultOptionComparatorIsNotNull()
    {
        HelpFormatter hf = new HelpFormatter();

        assertNotNull(hf.getOptionComparator());
    }

    @Test
    @SuppressWarnings("unchecked")
    public void testDefaultOptionComparatorIsCaseInsensitive()
    {
        HelpFormatter hf = new HelpFormatter();
        Comparator<Object> cmp = hf.getOptionComparator();

        assertTrue(cmp.compare(new Option("a", "x"), new Option("B", "x")) < 0);
        assertTrue(cmp.compare(new Option("B", "x"), new Option("a", "x")) > 0);
        assertEquals(0, cmp.compare(new Option("a", "x"), new Option("A", "x")));
    }

    @Test
    public void testDefaultOrderingInUsageIgnoresCase()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options()
                .addOption("B", false, "second")
                .addOption("a", false, "first")
                .addOption("c", false, "third");

        String out = run(pw -> hf.printUsage(pw, 80, "app", options));

        assertEquals("usage: app [-a] [-B] [-c]" + EOL, out);
    }

    @Test
    @SuppressWarnings("unchecked")
    public void testSetOptionComparatorNullRestoresDefaultComparator()
    {
        HelpFormatter hf = new HelpFormatter();
        Comparator<Object> reverse = (o1, o2) ->
                ((Option) o2).getKey().compareToIgnoreCase(((Option) o1).getKey());
        hf.setOptionComparator(reverse);
        assertTrue(hf.getOptionComparator() == reverse);

        hf.setOptionComparator(null);

        assertNotNull(hf.getOptionComparator());
        assertNotSame(reverse, hf.getOptionComparator());
        Options options = new Options()
                .addOption("a", false, "first")
                .addOption("B", false, "second")
                .addOption("c", false, "third");
        String out = run(pw -> hf.printUsage(pw, 80, "app", options));
        assertEquals("usage: app [-a] [-B] [-c]" + EOL, out);
    }

    // ---------------------------------------------------------------- Req. 8

    @Test
    public void testPrintOptionsWritesRenderedOptionsWithTrailingNewLine()
    {
        HelpFormatter hf = new HelpFormatter();
        Options options = new Options().addOption("a", false, "first");

        String out = run(pw -> hf.printOptions(pw, 80, options, 1, 3));

        assertEquals(" -a   first" + EOL, out);
    }

    @Test
    public void testPrintWrappedWritesTextWithTrailingNewLine()
    {
        HelpFormatter hf = new HelpFormatter();

        String out = run(pw -> hf.printWrapped(pw, 80, "hello world"));

        assertEquals("hello world" + EOL, out);
    }

    @Test
    public void testPrintWrappedWrapsTextAtWidth()
    {
        HelpFormatter hf = new HelpFormatter();

        String out = run(pw -> hf.printWrapped(pw, 10, "aaaa bbbb cccc"));

        assertEquals("aaaa bbbb" + EOL + "cccc" + EOL, out);
    }
}
