    /** Devuelve true si el nombre de SO del sistema empieza por el prefijo indicado. */
    private static boolean osNameStartsWith(String prefix) {
        String osName = System.getProperty("os.name");
        return osName != null && osName.startsWith(prefix);
    }

    /** Extrae el número mayor de una versión ya recortada, por ejemplo "11.0.2" -> 11. */
    private static int majorOf(String trimmedVersion) {
        int end = 0;
        while (end < trimmedVersion.length() && Character.isDigit(trimmedVersion.charAt(end))) {
            end++;
        }
        return Integer.parseInt(trimmedVersion.substring(0, end));
    }

    @Test
    public void javaVersionTrimmedIsSuffixOfJavaVersionStartingWithDigit() {
        String version = SystemUtils.JAVA_VERSION;
        String trimmed = SystemUtils.JAVA_VERSION_TRIMMED;
        if (version == null) {
            Assert.assertNull("Sin JAVA_VERSION, el valor recortado debe ser null", trimmed);
            return;
        }
        boolean hasDigit = false;
        for (int i = 0; i < version.length(); i++) {
            if (Character.isDigit(version.charAt(i))) {
                hasDigit = true;
                break;
            }
        }
        if (hasDigit) {
            Assert.assertNotNull(trimmed);
            Assert.assertTrue("Debe empezar por un dígito", Character.isDigit(trimmed.charAt(0)));
            Assert.assertTrue("Debe ser un sufijo de JAVA_VERSION", version.endsWith(trimmed));
        }
    }

    @Test
    public void javaVersionTrimmedHasNoLeadingLetters() {
        String trimmed = SystemUtils.JAVA_VERSION_TRIMMED;
        if (trimmed != null && !trimmed.isEmpty()) {
            Assert.assertFalse("No debe empezar por letra", Character.isLetter(trimmed.charAt(0)));
        }
    }

    @Test
    public void javaVersionIntIsPositiveWhenVersionIsKnown() {
        if (SystemUtils.JAVA_VERSION_TRIMMED != null) {
            Assert.assertTrue("JAVA_VERSION_INT debe ser positivo", SystemUtils.JAVA_VERSION_INT > 0);
            Assert.assertTrue("JAVA_VERSION_FLOAT debe ser positivo", SystemUtils.JAVA_VERSION_FLOAT > 0f);
        }
    }

    @Test
    public void javaVersionIntReflectsMajorVersionForModernJava() {
        String trimmed = SystemUtils.JAVA_VERSION_TRIMMED;
        if (trimmed != null) {
            int major = majorOf(trimmed);
            if (major >= 9) {
                Assert.assertTrue("Para Java " + major + " JAVA_VERSION_INT debe ser >= " + (major * 100),
                        SystemUtils.JAVA_VERSION_INT >= major * 100);
                Assert.assertTrue("Para Java " + major + " JAVA_VERSION_FLOAT debe ser >= " + major,
                        SystemUtils.JAVA_VERSION_FLOAT >= major);
            }
        }
    }

    @Test
    public void javaVersionIntIs180ForJava8() {
        String trimmed = SystemUtils.JAVA_VERSION_TRIMMED;
        if (trimmed != null && trimmed.startsWith("1.8.")) {
            Assert.assertEquals(180, SystemUtils.JAVA_VERSION_INT);
            Assert.assertEquals(1.8f, SystemUtils.JAVA_VERSION_FLOAT, 0.000001f);
        }
    }

    @Test
    public void isJavaVersionAtLeastIntAgreesWithFloatVariant() {
        boolean intResult = SystemUtils.isJavaVersionAtLeast(180);
        boolean floatResult = SystemUtils.isJavaVersionAtLeast(1.8f);
        Assert.assertEquals("Para la versión 1.8, ambas variantes deben coincidir si la versión actual es >= 1.8",
                SystemUtils.JAVA_VERSION_FLOAT >= 1.8f, floatResult);
        Assert.assertEquals(SystemUtils.JAVA_VERSION_INT >= 180, intResult);
    }

    @Test
    public void atMostOneLegacyJavaFlagIsTrue() {
        int count = 0;
        if (SystemUtils.IS_JAVA_1_1) count++;
        if (SystemUtils.IS_JAVA_1_2) count++;
        if (SystemUtils.IS_JAVA_1_3) count++;
        if (SystemUtils.IS_JAVA_1_4) count++;
        if (SystemUtils.IS_JAVA_1_5) count++;
        if (SystemUtils.IS_JAVA_1_6) count++;
        if (SystemUtils.IS_JAVA_1_7) count++;
        Assert.assertTrue("Como máximo una bandera IS_JAVA_1_x puede ser true", count <= 1);
    }

    @Test
