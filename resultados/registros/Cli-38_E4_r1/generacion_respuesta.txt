    private static Options basicOptions() {

        return options;

    /** Parsea un único flag "v" desde Properties y devuelve si quedó presente. */
    private static boolean flagFromProperty(String value) throws ParseException {
        Options options = new Options();
        options.addOption("v", false, "verbose");
        Properties properties = new Properties();
        properties.setProperty("v", value);
        CommandLine cmd = new DefaultParser().parse(options, new String[0], properties);
        return cmd.hasOption("v");
    }

    @Test
    public void stopAtNonOptionTrueAddsAllRemainingTokensAsArgs() throws ParseException {
        DefaultParser parser = new DefaultParser();
        CommandLine cmd = parser.parse(basicOptions(),
                new String[] {"-a", "file1", "-b", "x"}, true);

        assertTrue("La opción -a anterior al no-opción debe procesarse", cmd.hasOption("a"));
        assertFalse("-b tras el no-opción no debe procesarse como opción", cmd.hasOption("b"));
        assertEquals(Arrays.asList("file1", "-b", "x"), cmd.getArgList());
    }

    @Test
    public void stopAtNonOptionTrueTreatsUnknownDashTokenAfterStopAsArgument() throws ParseException {
        DefaultParser parser = new DefaultParser();
        CommandLine cmd = parser.parse(basicOptions(),
                new String[] {"file1", "-z"}, true);

        assertEquals(Arrays.asList("file1", "-z"), cmd.getArgList());
        assertFalse(cmd.hasOption("a"));
        assertFalse(cmd.hasOption("b"));
    }

    @Test
    public void stopAtNonOptionTrueWithOnlyNonOptionsLeavesNoOptions() throws ParseException {
        DefaultParser parser = new DefaultParser();
        CommandLine cmd = parser.parse(basicOptions(),
                new String[] {"-a", "uno"}, true);

        assertTrue(cmd.hasOption("a"));
        assertEquals(Arrays.asList("uno"), cmd.getArgList());
    }

    public void propertyValuesYesTrueOneActivateNoArgOptionIgnoringCase() throws ParseException {

    @Test
    public void propertyValuesOtherThanYesTrueOneDoNotActivateNoArgOption() throws ParseException {
        String[] nonActivators = {"no", "false", "0", "", "sí", "2"};

            assertFalse("El valor '" + value + "' no debe activar la opción", flagFromProperty(value));
        }

        options.addOptionGroup(group);

        properties.setProperty("b", "true");

        DefaultParser parser = new DefaultParser();
