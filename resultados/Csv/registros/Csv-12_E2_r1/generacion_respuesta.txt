    private static void assertNotEqualFormats(final CSVFormat a, final CSVFormat b) {
        assertFalse(a.equals(b));

    // ---------------------------------------------------------------- newFormat

    @Test
    public void testNewFormatInitializesAllOptionalFieldsToDefaults() {
        final CSVFormat format = CSVFormat.newFormat(';');

        assertEquals(';', format.getDelimiter());
        assertNull(format.getQuoteCharacter());
        assertNull(format.getQuoteMode());
        assertNull(format.getCommentMarker());
        assertNull(format.getEscapeCharacter());
        assertNull(format.getNullString());
        assertNull(format.getRecordSeparator());
        assertNull(format.getHeader());
        assertFalse(format.getIgnoreEmptyLines());
        assertFalse(format.getIgnoreSurroundingSpaces());
        assertFalse(format.getAllowMissingColumnNames());
        assertFalse(format.getSkipHeaderRecord());
    }

    @Test(expected = IllegalArgumentException.class)
    public void testNewFormatWithLineFeedDelimiterThrows() {
        CSVFormat.newFormat('\n');
    }

    @Test(expected = IllegalArgumentException.class)
    public void testNewFormatWithCarriageReturnDelimiterThrows() {
        CSVFormat.newFormat('\r');
    }

    @Test
    public void testWithEscapeNullDisablesEscaping() {

    @Test

    @Test

    @Test

    @Test

    @Test
    public void testEqualsDetectsEscapeNullVersusSet() {

    @Test
    public void testParseReturnsRecordsWithConfiguredDelimiter() throws Exception {
        try (CSVParser parser = CSVFormat.newFormat(';').parse(new StringReader("a;b\r\nc;d\r\n"))) {
            final List<CSVRecord> records = parser.getRecords();

            assertEquals(2, records.size());

        assertEquals("a,b\r\n", sb.toString());

    // ------------------------------------------------------- serialization

    @Test
    public void testSerializationRoundTripPreservesFullyConfiguredFormat() throws Exception {
        final CSVFormat original = CSVFormat.DEFAULT.withCommentMarker('#')
                .withEscape('\\')
                .withNullString("NULL")
                .withHeader("h1", "h2")
                .withQuoteMode(QuoteMode.ALL)
                .withSkipHeaderRecord(true);

        final ByteArrayOutputStream out = new ByteArrayOutputStream();
        try (ObjectOutputStream oos = new ObjectOutputStream(out)) {
            oos.writeObject(original);
        }

        final CSVFormat restored;
        try (ObjectInputStream in = new ObjectInputStream(new ByteArrayInputStream(out.toByteArray()))) {
            restored = (CSVFormat) in.readObject();
        }

        assertNotNull(restored);
        assertEquals(original, restored);
        assertEquals(original.hashCode(), restored.hashCode());
        assertArrayEquals(new String[] {"h1", "h2"}, restored.getHeader());
        assertEquals("NULL", restored.getNullString());
        assertEquals(QuoteMode.ALL, restored.getQuoteMode());
        assertTrue(restored.getSkipHeaderRecord());
    }

Nota: los números de línea del bloque E2 no coinciden con el código fuente mostrado (p. ej., la línea 863 no existe en esa versión), por lo que las pruebas apuntan a las ramas no cubiertas por comportamiento: validación en constructor, `with*(null)`, rama de duplicados con `null`, `equals`/`hashCode` con campos nulos frente a definidos, `toString` según opciones, formateo con valores nulos y comillas, y serialización con cabecera.
