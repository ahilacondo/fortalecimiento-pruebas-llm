/**

public class CSVFormatLLMTest {

    private enum Color {
        RED, GREEN
    }

    private static List<CSVRecord> parse(final CSVFormat format, final String input) throws IOException {
        try (CSVParser parser = format.parse(new StringReader(input))) {
            return parser.getRecords();
        }
    }

    // ---------------------------------------------------------------------
    // Formatos predefinidos

    @Test
    public void testPredefinedEnumReturnsMatchingFormat() {
        assertSame(CSVFormat.EXCEL, CSVFormat.Predefined.Excel.getFormat());
        assertSame(CSVFormat.MYSQL, CSVFormat.Predefined.MySQL.getFormat());
        assertSame(CSVFormat.RFC4180, CSVFormat.Predefined.RFC4180.getFormat());
        assertSame(CSVFormat.TDF, CSVFormat.Predefined.TDF.getFormat());
    }

    @Test
    public void testPredefinedValuesAllHaveFormats() {
        for (final CSVFormat.Predefined predefined : CSVFormat.Predefined.values()) {
            assertNotNull("Formato nulo para " + predefined.name(), predefined.getFormat());
        }
    }

    @Test
    public void testValueOfReturnsPredefinedFormat() {
        assertSame(CSVFormat.MYSQL, CSVFormat.valueOf("MySQL"));
        assertSame(CSVFormat.DEFAULT, CSVFormat.valueOf("Default"));
    }

    @Test(expected = IllegalArgumentException.class)
    public void testValueOfUnknownNameThrowsException() {
        CSVFormat.valueOf("NoExisteEsteFormato");
    }

    @Test
    public void testDefaultFormatSettings() {
        final CSVFormat format = CSVFormat.DEFAULT;
        assertEquals(',', format.getDelimiter());
        assertEquals(Character.valueOf('"'), format.getQuoteCharacter());
        assertEquals("\r\n", format.getRecordSeparator());
        assertTrue(format.getIgnoreEmptyLines());
        assertFalse(format.getIgnoreSurroundingSpaces());
        assertFalse(format.getAllowMissingColumnNames());
        assertNull(format.getCommentMarker());
        assertNull(format.getEscapeCharacter());
        assertNull(format.getNullString());
    }

    @Test
    public void testExcelFormatSettings() {
        final CSVFormat format = CSVFormat.EXCEL;
        assertEquals(',', format.getDelimiter());
        assertEquals("\r\n", format.getRecordSeparator());
        assertFalse(format.getIgnoreEmptyLines());
        assertTrue(format.getAllowMissingColumnNames());
    }

    @Test
    public void testRfc4180IgnoresEmptyLinesOnlyDifferenceFromDefault() {
        assertFalse(CSVFormat.RFC4180.getIgnoreEmptyLines());
        assertEquals(CSVFormat.DEFAULT.getDelimiter(), CSVFormat.RFC4180.getDelimiter());
        assertEquals(CSVFormat.DEFAULT.getRecordSeparator(), CSVFormat.RFC4180.getRecordSeparator());
    }

    @Test
    public void testTdfFormatSettings() {
        final CSVFormat format = CSVFormat.TDF;
        assertEquals('\t', format.getDelimiter());
        assertTrue(format.getIgnoreSurroundingSpaces());
        assertEquals(Character.valueOf('"'), format.getQuoteCharacter());
    }

    @Test
    public void testMySqlFormatSettings() {
        final CSVFormat format = CSVFormat.MYSQL;
        assertEquals('\t', format.getDelimiter());
        assertEquals(Character.valueOf('\\'), format.getEscapeCharacter());
        assertNull(format.getQuoteCharacter());
        assertEquals("\n", format.getRecordSeparator());
        assertEquals("\\N", format.getNullString());
        assertEquals(QuoteMode.ALL_NON_NULL, format.getQuoteMode());
        assertFalse(format.getIgnoreEmptyLines());
    }

    @Test
    public void testInformixUnloadFormatSettings() {
        final CSVFormat format = CSVFormat.INFORMIX_UNLOAD;
        assertEquals('|', format.getDelimiter());
        assertEquals(Character.valueOf('\\'), format.getEscapeCharacter());
        assertEquals("\n", format.getRecordSeparator());
    }

    @Test
    public void testPostgreSqlCsvFormatSettings() {
        final CSVFormat format = CSVFormat.POSTGRESQL_CSV;
        assertEquals(',', format.getDelimiter());
        assertEquals(Character.valueOf('"'), format.getQuoteCharacter());
        assertEquals(Character.valueOf('"'), format.getEscapeCharacter());
        assertEquals("", format.getNullString());
        assertEquals(QuoteMode.ALL_NON_NULL, format.getQuoteMode());
    }

    @Test
    public void testWithMethodsReturnNewInstanceAndKeepOriginal() {
        final CSVFormat modified = CSVFormat.DEFAULT.withDelimiter(';');
        assertNotSame(CSVFormat.DEFAULT, modified);
        assertEquals(';', modified.getDelimiter());
        assertEquals(',', CSVFormat.DEFAULT.getDelimiter());
    }

    @Test
    public void testWithQuoteNullDoesNotAlterPredefinedFormat() {
        final CSVFormat noQuote = CSVFormat.DEFAULT.withQuote(null);
        assertFalse(noQuote.isQuoteCharacterSet());
        assertTrue(CSVFormat.DEFAULT.isQuoteCharacterSet());
    }

    @Test
    public void testNewFormatHasOnlyDelimiterConfigured() {
        final CSVFormat format = CSVFormat.newFormat('X');
        assertEquals('X', format.getDelimiter());
        assertFalse(format.isQuoteCharacterSet());
        assertFalse(format.isEscapeCharacterSet());
        assertFalse(format.isCommentMarkerSet());
        assertFalse(format.isNullStringSet());
        assertFalse(format.getIgnoreEmptyLines());
        assertNull(format.getRecordSeparator());
        assertNull(format.getHeader());
    }

    @Test
    public void testNewFormatWithLineBreakDelimiterThrowsException() {

    public void testWithNullStringNullDisablesNullString() {

    @Test
    public void testSerializationPreservesExcelSettings() throws Exception {
        final ByteArrayOutputStream bytes = new ByteArrayOutputStream();
        try (ObjectOutputStream oos = new ObjectOutputStream(bytes)) {
            oos.writeObject(CSVFormat.EXCEL);
        }
        try (ObjectInputStream ois = new ObjectInputStream(new ByteArrayInputStream(bytes.toByteArray()))) {
            final CSVFormat restored = (CSVFormat) ois.readObject();
            assertEquals(CSVFormat.EXCEL, restored);
            assertTrue(restored.getAllowMissingColumnNames());
            assertFalse(restored.getIgnoreEmptyLines());
        }
    }

    @Test
    public void testPrintRecordAppendsRecordSeparator() throws Exception {
        final StringBuilder sb = new StringBuilder();
        CSVFormat.DEFAULT.printRecord(sb, "x", 1, true);
        assertEquals("x,1,true\r\n", sb.toString());
    }

    @Test
    public void testPrintlnWithoutTrailingDelimiterOutputsOnlySeparator() throws Exception {
        final StringBuilder sb = new StringBuilder();
        CSVFormat.DEFAULT.println(sb);
        assertEquals("\r\n", sb.toString());
    }

    @Test
    public void testPrintlnWithTrailingDelimiterOutputsDelimiterThenSeparator() throws Exception {
        final StringBuilder sb = new StringBuilder();
        CSVFormat.DEFAULT.withTrailingDelimiter().println(sb);
        assertEquals(",\r\n", sb.toString());
    }

    @Test
    public void testPrintObjectWritesValueWithoutLeadingDelimiterWhenNewRecord() throws Exception {
        final StringBuilder sb = new StringBuilder();
        CSVFormat.DEFAULT.print("valor", sb, true);
        assertEquals("valor", sb.toString());
    }

    @Test
    public void testPrintObjectWritesLeadingDelimiterWhenNotNewRecord() throws Exception {
        final StringBuilder sb = new StringBuilder("previo");
        CSVFormat.DEFAULT.print("valor", sb, false);
        assertEquals("previo,valor", sb.toString());
    }
