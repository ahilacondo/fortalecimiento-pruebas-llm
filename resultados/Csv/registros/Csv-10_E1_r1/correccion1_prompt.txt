/**
 * Pruebas adicionales para {@link CSVPrinter} que usan únicamente su API pública.
 */
public class CSVPrinterLLMTest {

    private static final String RS = CSVFormat.DEFAULT.getRecordSeparator();

    /** Appendable que registra las llamadas a flush y close. */
    private static final class TrackingAppendable implements Appendable, Flushable, Closeable {
        private final StringBuilder buffer = new StringBuilder();
        private boolean flushed;
        private boolean closed;

        @Override
        public Appendable append(final CharSequence csq) {
            buffer.append(csq);
            return this;
        }

        @Override
        public Appendable append(final CharSequence csq, final int start, final int end) {
            buffer.append(csq, start, end);
            return this;
        }

        @Override
        public Appendable append(final char c) {
            buffer.append(c);
            return this;
        }

        @Override
        public void flush() {
            flushed = true;
        }

        @Override
        public void close() {
            closed = true;
        }
    }

    /** Imprime un registro con los valores dados y devuelve el texto producido. */
    private static String printRecordWith(final CSVFormat format, final Object... values) throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, format);
        printer.printRecord(values);
        printer.close();
        return sw.toString();
    }

    @Test
    public void testPrintRecordEmptyIterableWritesOnlyRecordSeparator() throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, CSVFormat.DEFAULT);
        printer.printRecord(new ArrayList<String>());
        printer.close();
        assertEquals(RS, sw.toString());
    }

    @Test
    public void testPrintRecordIterableMatchesVarargsOutput() throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, CSVFormat.DEFAULT);
        printer.printRecord(Arrays.asList("a", "b,c"));
        printer.close();
        assertEquals("a,\"b,c\"" + RS, sw.toString());
    }

    @Test
    public void testConstructorDoesNotWriteAnything() throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, CSVFormat.DEFAULT);
        assertEquals("", sw.toString());
        printer.close();
    }

    @Test
    public void testGetOutReturnsSameAppendable() throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, CSVFormat.DEFAULT);
        assertSame(sw, printer.getOut());
        printer.close();
    }

    @Test
    public void testCustomRecordSeparatorIsUsed() throws IOException {
        final StringWriter sw = new StringWriter();
        final CSVPrinter printer = new CSVPrinter(sw, CSVFormat.DEFAULT.withRecordSeparator("\n"));
        printer.printRecord("a", "b");
        printer.close();
        assertEquals("a,b\n", sw.toString());
    }

    @Test
    public void testCustomDelimiterDoesNotQuoteComma() throws IOException {
        assertEquals("a,b;c" + RS,
                printRecordWith(CSVFormat.DEFAULT.withDelimiter(';'), "a,b", "c"));
    }

    @Test
    public void testCustomDelimiterQuotesValueContainingDelimiter() throws IOException {
        assertEquals("\"a;b\";c" + RS,
                printRecordWith(CSVFormat.DEFAULT.withDelimiter(';'), "a;b", "c"));
    }

    @Test
    public void testTabDelimiterDoesNotQuoteComma() throws IOException {
        assertEquals("a,b\tc" + RS,
                printRecordWith(CSVFormat.DEFAULT.withDelimiter('\t'), "a,b", "c"));
    }

    @Test
    public void testLeadingSpaceIsQuoted() throws IOException {
        assertEquals("\" a\"" + RS, printRecordWith(CSVFormat.DEFAULT, " a"));
    }

    @Test
    public void testLeadingSpaceInMiddleOfRecordIsQuoted() throws IOException {
        assertEquals("x,\" a\"" + RS, printRecordWith(CSVFormat.DEFAULT, "x", " a"));
    }

    @Test
    public void testLeadingHashIsQuoted() throws IOException {
        assertEquals("\"#x\"" + RS, printRecordWith(CSVFormat.DEFAULT, "#x"));
    }

    @Test
    public void testTrailingControlCharacterIsQuoted() throws IOException {
        assertEquals("\"a\u0001\"" + RS, printRecordWith(CSVFormat.DEFAULT, "a\u0001"));
    }

    @Test
    public void testEmptyFirstValueIsQuoted() throws IOException {
        assertEquals("\"\",x" + RS, printRecordWith(CSVFormat.DEFAULT, "", "x"));
    }

    @Test
    public void testEmptyMiddleValueIsNotQuoted() throws IOException {
        assertEquals("a,,b" + RS, printRecordWith(CSVFormat.DEFAULT, "a", "", "b"));
    }

    @Test
    public void testSingleEmptyValueRecordIsQuoted() throws IOException {
        assertEquals("\"\"" + RS, printRecordWith(CSVFormat.DEFAULT, ""));
    }

    public void testNullFirstValueIsWrittenAsQuotedEmpty() throws IOException {

        printer.print(null);
        printer.close();
        assertEquals("\"\"", sw.toString());
    }
