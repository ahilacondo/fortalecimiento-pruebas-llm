    /**
     * Construye una representación textual de todos los ajustes públicos del formato,
     * para comparar el estado antes y después de una operación.
     */
    private static String snapshot(final CSVFormat f) {
        return "delim=" + f.getDelimiter()
                + ";quote=" + f.getQuoteCharacter()
                + ";quoteMode=" + f.getQuoteMode()
                + ";comment=" + f.getCommentMarker()
                + ";escape=" + f.getEscapeCharacter()
                + ";nullString=" + f.getNullString()
                + ";recSep=" + f.getRecordSeparator()
                + ";header=" + Arrays.toString(f.getHeader())
                + ";headerComments=" + Arrays.toString(f.getHeaderComments())
                + ";ignoreEmpty=" + f.getIgnoreEmptyLines()
                + ";ignoreSpaces=" + f.getIgnoreSurroundingSpaces()
                + ";ignoreHeaderCase=" + f.getIgnoreHeaderCase()
                + ";skipHeader=" + f.getSkipHeaderRecord()
                + ";allowMissing=" + f.getAllowMissingColumnNames()
                + ";trim=" + f.getTrim()
                + ";trailingDelim=" + f.getTrailingDelimiter()
                + ";autoFlush=" + f.getAutoFlush();
    }

    /**
     * Verifica que una operación "with" devuelve una instancia nueva, que el original
     * no cambia y que el resultado refleja un cambio de estado.
     */
    private static void assertReturnsNewInstanceAndKeepsOriginal(final String label, final CSVFormat base,
            final Function<CSVFormat, CSVFormat> op) {
        final String before = snapshot(base);
        final CSVFormat derived = op.apply(base);
        assertNotNull(label + ": el resultado no debe ser null", derived);
        assertNotSame(label + ": debe devolver una instancia nueva", base, derived);
        assertEquals(label + ": el formato original no debe modificarse", before, snapshot(base));
        assertFalse(label + ": el resultado debe reflejar el cambio solicitado", before.equals(snapshot(derived)));
    }
