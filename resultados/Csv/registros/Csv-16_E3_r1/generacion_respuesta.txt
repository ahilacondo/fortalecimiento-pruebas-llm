public class CSVParserLLMTest {

    // 1. parse(InputStream, Charset, CSVFormat): lectura real y charset

    @Test
    public void testParseInputStreamDecodesUtf8WithGivenCharset() throws IOException {
        final byte[] data = "\u00e9,\u00f1\n1,2\n".getBytes(StandardCharsets.UTF_8);
        try (CSVParser parser = CSVParser.parse(new ByteArrayInputStream(data), StandardCharsets.UTF_8,
                CSVFormat.DEFAULT)) {
            final List<CSVRecord> records = parser.getRecords();
            assertEquals(2, records.size());
            assertArrayEquals(new String[] { "\u00e9", "\u00f1" }, records.get(0).values());
            assertArrayEquals(new String[] { "1", "2" }, records.get(1).values());
        }
    }
